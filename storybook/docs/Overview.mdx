export const managerHref = (link) => {
  // Storybook vit dans la fenetre du haut, cette page de doc dans son iframe
  // `/iframe.html`. Une URL relative se resout contre le document qui la
  // porte, donc contre `/iframe.html` et non contre la racine, y compris dans
  // un `<a target="_top">` : un `?path=` relatif devenait `/iframe.html?path=`,
  // la fenetre du haut chargeait la coquille de preview, et la doc s'affichait
  // sans le menu ni la barre d'outils. Il faut donc un lien absolu.
  //
  // Le pathname prive de son dernier segment donne la racine du Storybook :
  // `/` reste `/`, `/iframe.html` redevient `/`, et un deploiement
  // sous-chemin (`/storybook/`) est conserve.
  const top = window.top.location;
  const root = top.pathname.slice(0, top.pathname.lastIndexOf('/') + 1);
  return `${top.origin}${root}${link}`;
};

export const ComponentCard = ({ name, description, link, story, previewHeight = 154, previewScale = 0.9 }) => {
  const handleCardClick = (e) => {
    // La preview est interactive, et le lien navigue deja tout seul.
    const previewArea = e.currentTarget.querySelector('[data-preview-area]');
    if (previewArea && previewArea.contains(e.target)) return;
    if (e.target.closest('a')) return;

    window.top.location.href = managerHref(link);
  };

  return (
    <div
      style={{
        display: 'flex', flexDirection: 'column', padding: 20, borderRadius: 12,
        background: 'var(--sb-bg-subtle)', border: '1px solid var(--sb-border)',
        transition: 'all .2s ease', cursor: 'pointer'
      }}
      onClick={handleCardClick}
      onMouseOver={(e) => { e.currentTarget.style.borderColor = 'var(--sb-accent)'; e.currentTarget.style.transform = 'translateY(-2px)'; e.currentTarget.style.boxShadow = '0 4px 12px rgba(0,0,0,.1)'; }}
      onMouseOut={(e) => { e.currentTarget.style.borderColor = 'var(--sb-border)'; e.currentTarget.style.transform = 'translateY(0)'; e.currentTarget.style.boxShadow = 'none'; }}
    >
      <div
        data-preview-area="true"
